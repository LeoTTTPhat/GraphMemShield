% =====================================================================
% Threat-model figure: cross-session contamination on a tiny KG
% =====================================================================
\begin{tikzpicture}[
    >=Latex,
    font=\footnotesize,
    every node/.style={inner sep=1.5pt},
    ent/.style={circle, draw=black, fill=white, minimum size=0.55cm},
    sess/.style={ellipse, draw, dashed, inner sep=4pt, align=center,
                 font=\scriptsize\itshape},
    relV/.style={->, thick, red!70!black},
    relA/.style={->, thick, blue!60!black},
    leak/.style={->, very thick, orange!85!black, decorate,
                 decoration={snake, amplitude=.5mm, segment length=2mm}}
]
% --- entity nodes ------------------------------------------------------
\node[ent] (alice)   at (-3.0, 1.2) {\scriptsize A};
\node[ent] (clinic)  at (-1.0, 1.2) {\scriptsize C};
\node[ent] (cond)    at ( 1.0, 1.2) {\scriptsize $\kappa$};
\node[ent] (bob)     at (-3.0,-1.2) {\scriptsize B};
\node[ent] (gym)     at (-1.0,-1.2) {\scriptsize G};
\node[ent] (laptop)  at ( 1.0,-1.2) {\scriptsize L};
\node[ent] (req)     at ( 3.4, 0.0) {\scriptsize $u_r$};
% --- victim session ----------------------------------------------------
\draw[relV] (alice)  -- node[above,font=\tiny\itshape]{visited}    (clinic);
\draw[relV] (clinic) -- node[above,font=\tiny\itshape]{diagnosed}  (cond);
% --- another user / session -------------------------------------------
\draw[relA] (bob)    -- node[below,font=\tiny\itshape]{visited}    (gym);
\draw[relA] (bob)    .. controls +(0.6,-0.6) and +(-0.6,-0.6) ..
                      node[below,font=\tiny\itshape]{purchased}   (laptop);
% --- session boundaries -----------------------------------------------
\node[sess, fit={(alice)(clinic)(cond)},
      label={[red!60!black,font=\tiny]above:victim $s_v$ (medical)}] {};
\node[sess, fit={(bob)(gym)(laptop)},
      label={[blue!60!black,font=\tiny]below:other $s_o$ (financial)}] {};
% --- attacker probe ----------------------------------------------------
\draw[->, thick, dashed] (req) to[bend left=15]
      node[above,font=\tiny\itshape]{probe ``clinic''} (clinic);
% --- leakage path ------------------------------------------------------
\draw[leak] (clinic) to[bend right=20] (req);
\node[font=\tiny\itshape, orange!80!black]
      at (2.5, 0.85) {cross-session leak};
% --- legend ------------------------------------------------------------
\node[anchor=west, font=\tiny] at (-3.6,-2.2)
      {\textcolor{red!70!black}{$\rightarrow$} victim edge};
\node[anchor=west, font=\tiny] at (-1.0,-2.2)
      {\textcolor{blue!60!black}{$\rightarrow$} non-victim edge};
\node[anchor=west, font=\tiny] at ( 1.6,-2.2)
      {\textcolor{orange!85!black}{$\leadsto$} leakage};
\end{tikzpicture}
